cmake_minimum_required(VERSION 3.16)
project(quant_systems_lab VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(QSL_ENABLE_SANITIZERS "Enable AddressSanitizer and UBSan" OFF)

if(QSL_ENABLE_SANITIZERS)
    if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizers require GCC or Clang")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

add_library(qsl_book core/src/order_book.c)
target_include_directories(qsl_book PUBLIC core/include)
add_executable(book_demo examples/book_demo.c)
target_link_libraries(book_demo PRIVATE qsl_book)
add_executable(book_tests tests/test_order_book.c)
target_link_libraries(book_tests PRIVATE qsl_book)

foreach(target qsl_book book_demo book_tests)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()

enable_testing()
foreach(case scenario errors capacity limits)
    add_test(NAME book_${case} COMMAND book_tests ${case})
endforeach()
add_test(NAME book_demo_smoke COMMAND book_demo)
